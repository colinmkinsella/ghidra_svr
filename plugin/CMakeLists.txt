cmake_minimum_required(VERSION 3.24)
project(binja-ghidra CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC OFF)
set(CMAKE_AUTORCC OFF)

# ---------------------------------------------------------------------------
# Paths
# ---------------------------------------------------------------------------
set(BN_INSTALL_DIR "C:/Program Files/Vector35/BinaryNinja"
    CACHE PATH "Binary Ninja installation directory")

# ---------------------------------------------------------------------------
# Ghidra — download latest release from GitHub if GHIDRA_HOME is not set
# ---------------------------------------------------------------------------
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
include(FetchGhidra)

# Auto-generate bridge/gradle.properties so `./gradlew shadowJar` works
# without any manual setup after a fresh checkout.
if(GHIDRA_HOME AND NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../bridge/gradle.properties")
    file(WRITE "${CMAKE_CURRENT_SOURCE_DIR}/../bridge/gradle.properties"
        "# Auto-generated by CMake — edit or delete to override\n"
        "ghidraHome=${GHIDRA_HOME}\n")
    message(STATUS "Ghidra: wrote bridge/gradle.properties")
endif()

# ---------------------------------------------------------------------------
# Qt6 — find before FetchContent so FindBinaryNinjaUI picks up the cached result.
# FetchQtBuild clones Vector35/qt-build and registers a `build-qt` target for
# machines that don't have Qt installed yet.  It also sets Qt6_DIR to point at
# <repo-root>/qt/ if a previous build-qt run installed Qt there.
# ---------------------------------------------------------------------------
include(FetchQtBuild)

find_package(Qt6 QUIET COMPONENTS Widgets Core)
if(NOT Qt6_FOUND)
    # build-qt was added by FetchQtBuild above and is usable even if we stop here.
    # Using return() (not FATAL_ERROR) so the target makes it into the build files.
    if(TARGET build-qt)
        message(STATUS "")
        message(STATUS "Qt6 not found — plugin cannot be built yet.")
        message(STATUS "Step 1:  cmake --build plugin/build --target build-qt   (~1-2 hours)")
        message(STATUS "Step 2:  cmake -B plugin/build -S plugin                (re-configure)")
        message(STATUS "Step 3:  cmake --build plugin/build                     (build plugin)")
        message(STATUS "")
    else()
        message(FATAL_ERROR "Qt6 not found. Install Qt6 or pass -DQt6_DIR=<path>.")
    endif()
    return()
endif()
set_target_properties(Qt6::Core Qt6::Widgets PROPERTIES
    MAP_IMPORTED_CONFIG_RELWITHDEBINFO RELEASE
    MAP_IMPORTED_CONFIG_MINSIZEREL     RELEASE
)

# ---------------------------------------------------------------------------
# Binary Ninja API — auto-fetch the exact commit matching the BN install
# ---------------------------------------------------------------------------
include(FetchContent)

# On macOS the app bundle keeps api_REVISION.txt under Contents/Resources/,
# while BN_INSTALL_DIR points at Contents/MacOS/ (where the dylibs live).
if(APPLE)
    file(READ "${BN_INSTALL_DIR}/../Resources/api_REVISION.txt" _bn_rev_raw)
else()
    file(READ "${BN_INSTALL_DIR}/api_REVISION.txt" _bn_rev_raw)
endif()
# Split into lines, drop blanks, take the last one (the bare SHA-1).
string(REGEX MATCHALL "[^\r\n]+" _rev_lines "${_bn_rev_raw}")
list(FILTER _rev_lines EXCLUDE REGEX "^$")
list(GET _rev_lines -1 BN_API_COMMIT)
string(STRIP "${BN_API_COMMIT}" BN_API_COMMIT)
if(NOT BN_API_COMMIT MATCHES "^[0-9a-f]+$")
    message(FATAL_ERROR
        "Could not parse API commit from ${BN_INSTALL_DIR}/api_REVISION.txt\n"
        "Got: '${BN_API_COMMIT}'")
endif()
message(STATUS "Binary Ninja API commit: ${BN_API_COMMIT}")

FetchContent_Declare(binaryninja-api
    GIT_REPOSITORY https://github.com/Vector35/binaryninja-api.git
    GIT_TAG        ${BN_API_COMMIT}
    GIT_SHALLOW    TRUE
    GIT_SUBMODULES "vendor/fmt"
    SOURCE_DIR     "${CMAKE_SOURCE_DIR}/../binaryninja-api"
    SUBBUILD_DIR   "${CMAKE_BINARY_DIR}/_deps/binaryninja-api-subbuild"
)
FetchContent_MakeAvailable(binaryninja-api)


# ---------------------------------------------------------------------------
# Plugin target
# ---------------------------------------------------------------------------
add_library(binja-ghidra SHARED
    src/plugin.cpp
    src/BridgeProcess.cpp
    src/BridgeClient.cpp
    src/GhidraConnection.cpp
    src/SyncEngine.cpp
    src/ui/ConnectDialog.cpp
    src/ui/ProjectPanel.cpp
    # Headers with Q_OBJECT must be listed so AUTOMOC finds them.
    include/ui/ConnectDialog.h
    include/ui/ProjectPanel.h
)

target_include_directories(binja-ghidra PRIVATE
    include
    "${CMAKE_SOURCE_DIR}/../binaryninja-api/vendor"   # nlohmann/json.hpp
)

target_link_libraries(binja-ghidra PRIVATE
    binaryninjaui          # transitively pulls in binaryninjaapi + BN link dirs
    Qt6::Widgets
    Qt6::Core
)

if(WIN32)
    target_link_libraries(binja-ghidra PRIVATE ws2_32)
    set_target_properties(binja-ghidra PROPERTIES PREFIX "")
elseif(UNIX AND NOT APPLE)
    target_link_libraries(binja-ghidra PRIVATE dl)
endif()

target_compile_definitions(binja-ghidra PRIVATE
    NOMINMAX
    WIN32_LEAN_AND_MEAN
    _CRT_SECURE_NO_WARNINGS
)

# ---------------------------------------------------------------------------
# Install — uses bn_install_plugin() helper defined by the api
# ---------------------------------------------------------------------------
bn_install_plugin(binja-ghidra)
install(FILES
    "${CMAKE_SOURCE_DIR}/../bridge/build/libs/ghidra-bridge-0.1.0.jar"
    DESTINATION "${BN_USER_PLUGINS_DIR}"
    OPTIONAL
)
